# The C linter layer's committed suppressions (ci-debs R2.2, R2.3): the
# finding classes cppcheck must not report, each with the reason the
# class is excluded. One rule per line in cppcheck's `id:file[:line]`
# form, read by the Cppcheck function in .dagger/linters.go via
# --suppressions-list. The layer's contract is zero UNSUPPRESSED
# findings, so a rule may exist only for a class a real run reported,
# and the reason must state why the class is not a defect in this
# tree (or why its fix is deferred) - never "silence and see".

# vmlinux.h is the generated BTF dump of the running kernel's types
# (bpftool btf dump), ~140k lines of kernel ABI shapes, not authored
# code: any finding in it is noise by construction. Today's runs
# report nothing there; the rule states the policy should a future
# cppcheck model surface something in the dump.
*:bpf/vmlinux.h

# libbpf's helper shim, shipped by libbpf-dev: bpf_helper_defs.h is a
# generated file of static inline wrappers, each returning the BPF
# calling convention's __u64 cast back to the helper's real return
# type. One run reports 211 intToPointerCast findings, every one in
# this upstream header, none reachable from snitch's choices.
intToPointerCast:/usr/include/bpf/bpf_helper_defs.h

# bpf_map_lookup_elem returns map memory the verifier manages, not a
# stack or heap object with a C lifetime; cppcheck models only the
# latter, so returning a looked-up value reads to it as a local's
# address escaping (the three `if (v) return v;` lookup paths of
# snitch.bpf.c). The class is a modeling limit of any C analyzer on
# this idiom, so the rule covers the file, not the lines.
returnDanglingLifetime:bpf/snitch.bpf.c

# The map key/value structs are kernel ABI shapes mirrored to Go by
# the bpf2go -type flags in internal/bpf/gen.go: pad members hold
# layout alignment, name[] members are the key bytes a lookup passes
# raw (bpf_map_lookup_elem(&domains, w->buf + start)), dnshdr
# mirrors the wire header the parser stops reading after the
# question. No member can disappear without changing the bytes the
# kernel and userspace share, so member-usage analysis cannot make
# them dead; the rule covers each file's whole struct set.
unusedStructMember:bpf/snitch.bpf.c
unusedStructMember:bpf/dns.bpf.c

# the code fix is deferred; each line stands alone so any NEW finding

constVariablePointer:bpf/snitch.bpf.c:218
constVariablePointer:bpf/snitch.bpf.c:219
constVariablePointer:bpf/snitch.bpf.c:241

# pkt_init stores the skb pointer and returns 0, a shape cppcheck
# cannot infer value-flow for, so it assumes the constant return and
# marks the error branch's condition dead. The branches are live
# code any nonzero return of the initializer takes; the conditions
# are not decidable by the analyzer, per line.
knownConditionTrueFalse:bpf/dns.bpf.c:439
knownConditionTrueFalse:bpf/dns.bpf.c:514
